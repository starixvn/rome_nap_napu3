# create dictionary to contain sporttype, tote and type
${TypeDic}= Create Dictionary Super s4
Set To Dictionary ${TypeDic} Master m3
Set To Dictionary ${TypeDic} Agent a2
Set To Dictionary ${TypeDic} Member a1
# get sporttype, tote type and PT type

${Totetype}= Get From Dictionary ${ToteDict} ${Tote}

... ${type}
# Set value

Input Text ${tmpmax} ${max}
Input Text ${tmpmaxPM} ${maxPM}

Select From List By Label ${tmpcomm} ${comm}
Click Update ${tmplblName} ${ERBbtnUpdate} ${cfAccountPrefix}
${outputMsg} Get Edit Msg ${ERBdivMsg} ${cfTimeout} * 3

Expand Collapse RB Min PT [Arguments] ${Race} ${Tote} ${isExpand}
[Documentation] This action is used to Expand or Collapse RB Min PT.Just use it when current frame is Min PT Racing page\n
\n
\n
${isExpand}: 1 or 0 \n
\n
\n
${Race}: the RB type (HR,Greyhound or Harness)\n
\n
\n
${Tote}: SA, UK/Irish or Aust\n
\n
# get sporttype, tote type and PT type
${sporttype}= Get From Dictionary ${SporttypeDict} ${Race}
${Totetype}= Get From Dictionary ${ToteDict} ${Tote}
# Set xpath control
${tmplbSport} Set Variable //span[@target='setting_${sporttype}']
${tmplbTote} Set Variable //td[@target='setting_${sporttype}_${Totetype}']
# get status of Sport and Tote status of Sport = 'expanded expand' -> it is collapsed class of Tote contains md3 -> it is collapsed
${tmpSportClass}= Get Element Attribute ${tmplbSport}@class
${statusSport}= Set Variable If '${tmpSportClass}'== 'expanded expand' 0
... 1
${tmpToteClass}= Get Element Attribute ${tmplbTote}/following-sibling::td@class
${statusTote}= Get Substring ${tmpToteClass} -3
${statusTote}= Set Variable If '${statusTote}' == 'md3' 0
... 1
# Expand or collapse
Run Keyword If '${statusSport}'<>'${isExpand}' Click Element ${tmplbSport}
Run Keyword If '${statusTote}'<>'${isExpand}' Click Element ${tmplbTote}
