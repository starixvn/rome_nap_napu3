Get Sport Control [Arguments] ${type} ${sport} ${bettype}
[Documentation] This action is used to fill in sport bet setting (SB and NG)\n
\n
${type}: account type (Master/Agent/Member)\n
\n
${sport}: sporttype name \n
\n
${bettype}: bettype name\n
\n
${outputlist}: contains SportID,Sport label, Bet limit controls (minbet, maxbet, maxpermatch, maxperball), PT control (MaxPT, UplinePresetPT) and Comm control\n
\n
# Get sport bettye
${typelist}= Create List ${EMPTY} Member
... Agent Master Super
${tmptype}= Get Index From List ${typelist} ${type}
${tmpSportID}= Get From Dictionary ${SporttypeDict} ${sport}
${tmpbettypename}= Set Variable If ${tmpSportID} > 99 ${sport} ${bettype}
... ${bettype}
${tmpBettypetID}= Get From Dictionary ${BettypeDict} ${tmpbettypename}
# Get sport bettye
${tmpBettypetID}= Get From Dictionary ${BettypeDict} ${tmpbettypename}
# Set bet setting control
${Sportbetsetting}= Set Variable //div[@id='divBettingLimit']//div[text()='${sport}']/following-sibling::*
${tmpposMin} Set Variable if ${tmpSportID} == 161 5
... 1
${tmpStep} Set Variable if '${type}'=='Super' 1
... 2
${tmpposMax} Evaluate (int) (${tmpposMin} + ${tmpStep})
${tmpposMaxPM} Evaluate (int) (${tmpposMax}+ ${tmpStep})
${tmpposMaxPB} Evaluate ${tmpposMaxPM}+ ${tmpStep}
${tmpMin} Set Variable ${Sportbetsetting}[${tmpposMin}]/input
${tmpMax}= Set Variable ${Sportbetsetting}[${tmpposMax}]/input
${tmpMaxPM}= Set Variable ${Sportbetsetting}[${tmpposMaxPM}]/input
${tmpMaxPB}= Set Variable ${Sportbetsetting}[${tmpposMaxPB}]/input
# Set Comm control
${tmpComm}= Set Variable If ${tmpSportID} > 99 ${ESBcboComm${tmpSportID}}
... ${ESBcboComm1}
${tmpComm}= Set Variable If ${tmpSportID} <= 99 and '${type}'=='Member' ${ESBcboCommHDPMember}
... ${tmpComm}
# Set PT control
${tmpPrefixList}= Create List ${EMPTY} a
... a m s
... s
Comment ${tmPrefix}= Set Variable If '${type}'=='Member'
... a m
Comment ${tmPrefix}= Set Variable If '${type}'=='Super'
... s ${tmPrefix}
Comment ${tmPrefixUpline}= Set Variable If '${type}'=='Master'
... s m
${tmpTypeUpline}= Evaluate ${tmptype} + 1
${tmPrefix}= Get From List ${tmpPrefixList} ${tmptype}
${tmPrefixUpline}= Get From List ${tmpPrefixList} ${tmpTypeUpline}
${tmpUplinePreset}= Set Variable xpath=//select[@id='${tmPrefixUpline}${tmptype}\$${tmpSportID}\$${tmpBettypetID}\$0']
${tmpPT}= Set Variable xpath=//select[@id='${tmPrefix}${tmptype}\$${tmpSportID}\$${tmpBettypetID}\$0']
${tmplbSport}= Set Variable //div[@class='PTSport']/div[text()='${sport}']
# Return ouput list
${tmpMaxPB}= Set Variable If ${tmpSportID} == 161 ${tmpMaxPB}
... ${EMPTY}
${tmpUplinePreset}= Set Variable If '${type}'=='Super' ${EMPTY}
... ${tmpUplinePreset}
${outputlist}= Create List ${tmpSportID} ${tmplbSport}
... ${tmpMin} ${tmpMax} ${tmpMaxPM}
... ${tmpMaxPB} ${tmpPT} ${tmpUplinePreset}
... ${tmpComm}
[Return] ${outputlist}
